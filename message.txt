#!/bin/zsh

# Fix known NVM conflict
unset NPM_CONFIG_PREFIX

# Define paths
GOINFRE_DIR="$HOME/goinfre"
NVM_DIR="$GOINFRE_DIR/nvm"
PNPM_DIR="$GOINFRE_DIR/pnpm"
ZSHRC="$HOME/.zshrc"

# Create goinfre directory
mkdir -p "$GOINFRE_DIR"

# Clone NVM if not already present
if [ ! -d "$NVM_DIR" ]; then
  echo "Cloning NVM into $NVM_DIR..."
  git clone https://github.com/nvm-sh/nvm.git "$NVM_DIR"
else
  echo "✅ NVM already exists at $NVM_DIR"
fi

# Clean old NVM config from .zshrc if it exists
sed -i '' '/NVM config in goinfre/,+5d' "$ZSHRC"

# Add NVM config to .zshrc
echo "Updating $ZSHRC with NVM config..."
{
  echo ""
  echo "# >>> NVM config in goinfre >>>"
  echo "export NVM_DIR=\"$NVM_DIR\""
  echo "[ -s \"\$NVM_DIR/nvm.sh\" ] && \. \"\$NVM_DIR/nvm.sh\""
  echo "[ -s \"\$NVM_DIR/bash_completion\" ] && \. \"\$NVM_DIR/bash_completion\""
  echo "# <<< NVM config in goinfre <<<"
} >> "$ZSHRC"

# Source .zshrc to load NVM
echo "Sourcing $ZSHRC..."
source "$ZSHRC"

# Ensure the environment variable is gone
unset NPM_CONFIG_PREFIX

# Install Node.js LTS version if needed
echo "Installing Node.js LTS..."
nvm install --lts

# Set default and use it in current shell
nvm alias default node
nvm use node

# Show installed versions
echo "✅ Installed versions:"
which node
node -v
which npm
npm -v

# -----------------------------
# Install pnpm in goinfre
# -----------------------------
echo "Installing pnpm locally in $PNPM_DIR..."

curl -fsSL https://get.pnpm.io/install.sh | env PNPM_HOME="$PNPM_DIR" sh -

# Add pnpm to PATH in current session
export PNPM_HOME="$PNPM_DIR"
export PATH="$PNPM_HOME:$PATH"

# Clean old pnpm config from .zshrc if it exists
sed -i '' '/pnpm goinfre config/,+3d' "$ZSHRC"

# Add pnpm config to .zshrc
echo "Updating $ZSHRC with pnpm config..."
{
  echo ""
  echo "# >>> pnpm goinfre config >>>"
  echo "export PNPM_HOME=\"$PNPM_HOME\""
  echo "export PATH=\"\$PNPM_HOME:\$PATH\""
  echo "# <<< pnpm goinfre config <<<"
} >> "$ZSHRC"

# Confirm pnpm installation
if command -v pnpm >/dev/null 2>&1; then
  echo "✅ pnpm is installed:"
  which pnpm
  pnpm -v
else
  echo "❌ pnpm installation failed"
fi

echo "✅ NVM, Node.js, npm, and pnpm are fully working now."
